```c
void sort(SortList r)
{
    i = 1;
    while (i < N - i + 1)
    {
        min = max = i;
        for (j = i + 1; j <= N - i + 1; ++j)
            if (r[j].key < r[min].key) min = j;
            else if (r[j].key > r[max].key) max = j;
        if (min != i) { w = r[min]; r[min] = r[i]; r[i] = w; }
        if (max != N - i + 1)
        {
            if (max == i) { w = r[min]; r[min] = r[N - i + 1]; r[N - i + 1] = w; }
            else { w = r[max]; r[max] = r[N - i + 1]; r[N - i + 1] = w; }
        }
        i++;
    }
}
```
